              <TabItem label="Node.js">

```bash
npm install bunqueue-client
```

```javascript title="jobs.mjs"
import { Queue, Worker } from 'bunqueue-client';

const connection = { host: '127.0.0.1', port: 6789 };
const queue = new Queue('emails', connection);

const worker = new Worker(
  'emails',
  async (job) => {
    console.log('Processing:', job.data.to);
    return { sent: true };
  },
  connection
);

worker.on('error', (error) => console.error(error));
await queue.add('welcome', { to: 'hello@example.com' });
```

```bash
node jobs.mjs
```

              <TabItem label="Python">

```bash
pip install bunqueue-client
```

```python title="jobs.py"
from bunqueue import Queue, Worker

connection = {"host": "127.0.0.1", "port": 6789}
queue = Queue("emails", **connection)
queue.add("welcome", {"to": "hello@example.com"})
queue.close()

def process(job):
    print("Processing:", job.data["to"])
    return {"sent": True}

Worker("emails", process, **connection).run()
```

```bash
python jobs.py
```

          <div class="home-client-step">
            <h3>Add a job and process it</h3>

```typescript title="jobs.ts"
import { Queue, Worker } from 'bunqueue/client';

const queue = new Queue<{ to: string }>('emails', { embedded: true });

const worker = new Worker<{ to: string }>(
  'emails',
  async (job) => {
    console.log('Processing:', job.data.to);
    return { sent: true };
  },
  { embedded: true }
);

worker.on('error', (error) => console.error(error));
await queue.add('welcome', { to: 'hello@example.com' });
```

```bash
bun jobs.ts
```
